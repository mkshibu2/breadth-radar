fVolTxt(v) =>
    string out = ""
    if v >= 1e7
        out := str.tostring(math.round(v / 1e7 * 100) / 100) + "Cr"
    else if v >= 1e6
        out := str.tostring(math.round(v / 1e6 * 100) / 100) + "M"
    else if v >= 1e3
        out := str.tostring(math.round(v / 1e3 * 100) / 100) + "k"
    else
        out := str.tostring(v)
    out

//================= VOLUME TRIGGER V4 =================
// Simple Volume Profile based trigger for GTT placement.
//
// User selects the consolidation lookback.
// Volume is distributed across each candle's High-Low range.
// The dominant High Volume Node (HVN) is identified.
// The upper edge of the HVN is used as the Volume Trigger.
//
// Purpose:
// Help place a pending GTT above an important volume-acceptance
// area without requiring intraday RVol monitoring.
//
// This is NOT intended to replicate TradingView's full
// Volume Profile. It is a lightweight GTT-oriented tool.

//----------------- INPUTS -----------------

showVolTrigger = input.bool(false, "Show Volume Trigger Line")

vtLookback = input.int(
     25,
     "VT Lookback Bars",
     minval=10,
     maxval=100
     )

vtRows = input.int(
     30,
     "VT Price Rows",
     minval=10,
     maxval=60
     )

// Rows with volume >= this percentage of POC volume
// are considered part of the dominant HVN.
vtHvnPct = input.float(
     60.0,
     "VT HVN Threshold %",
     minval=40.0,
     maxval=90.0,
     step=5.0
     )

// A meaningful volume drop above the HVN.
// Used to confirm that price is moving from an
// accepted-volume area into a relatively thin area.
vtLvnPct = input.float(
     40.0,
     "VT LVN Threshold %",
     minval=10.0,
     maxval=60.0,
     step=5.0
     )

//----------------- PERSISTENT LINE -----------------

var line volTriggerLine = na

//----------------- MAIN CALCULATION -----------------

if barstate.islast

    // Always remove the previous line before recalculating.
    if not na(volTriggerLine)
        line.delete(volTriggerLine)
        volTriggerLine := na

    if showVolTrigger and isDaily and bar_index >= vtLookback

        //==================================================
        // 1. PRICE RANGE OF SELECTED LOOKBACK
        //==================================================

        vtHigh = ta.highest(high, vtLookback)
        vtLow  = ta.lowest(low, vtLookback)

        vtRange = vtHigh - vtLow

        if vtRange > 0

            rowSize = vtRange / vtRows

            // Volume accumulated at each price row.
            float[] vtVol =
                 array.new_float(vtRows, 0.0)

            //==================================================
            // 2. DISTRIBUTE EACH CANDLE'S VOLUME
            //    ACROSS ITS HIGH-LOW RANGE
            //==================================================

            for i = 0 to vtLookback - 1

                candleHigh = high[i]
                candleLow  = low[i]
                candleRange = candleHigh - candleLow

                if candleRange > 0 and not na(volume[i])

                    candleVol = volume[i]

                    // Determine which profile rows the candle
                    // overlaps.
                    firstRow =
                         int(
                             math.floor(
                                 (candleLow - vtLow) / rowSize
                             )
                         )

                    lastRow =
                         int(
                             math.floor(
                                 (candleHigh - vtLow) / rowSize
                             )
                         )

                    firstRow :=
                         math.max(
                             0,
                             math.min(vtRows - 1, firstRow)
                         )

                    lastRow :=
                         math.max(
                             0,
                             math.min(vtRows - 1, lastRow)
                         )

                    // Distribute volume according to the amount
                    // of the candle's price range occupying each row.
                    for r = firstRow to lastRow

                        rowLow =
                             vtLow + r * rowSize

                        rowHigh =
                             rowLow + rowSize

                        overlapLow =
                             math.max(candleLow, rowLow)

                        overlapHigh =
                             math.min(candleHigh, rowHigh)

                        overlap =
                             math.max(
                                 0.0,
                                 overlapHigh - overlapLow
                             )

                        if overlap > 0

                            allocatedVol =
                                 candleVol *
                                 overlap /
                                 candleRange

                            array.set(
                                 vtVol,
                                 r,
                                 array.get(vtVol, r) +
                                 allocatedVol
                             )

            //==================================================
            // 3. FIND POC / DOMINANT VOLUME ROW
            //==================================================

            float maxVol = 0.0
            int pocRow = 0

            for r = 0 to vtRows - 1

                rowVol =
                     array.get(vtVol, r)

                if rowVol > maxVol

                    maxVol := rowVol
                    pocRow := r

            if maxVol > 0

                //==================================================
                // 4. FIND THE DOMINANT HVN CLUSTER
                //==================================================

                hvnThreshold =
                     maxVol * vtHvnPct / 100.0

                int upperHvnRow = pocRow
                int lowerHvnRow = pocRow

                // Expand upward from POC.
                if pocRow < vtRows - 1

                    for r = pocRow + 1 to vtRows - 1

                        rowVol =
                             array.get(vtVol, r)

                        if rowVol >= hvnThreshold
                            upperHvnRow := r
                        else
                            break

                // Expand downward from POC.
                if pocRow > 0

                    for r = pocRow - 1 to 0

                        rowVol =
                             array.get(vtVol, r)

                        if rowVol >= hvnThreshold
                            lowerHvnRow := r
                        else
                            break

                //==================================================
                // 5. LOOK FOR THE LVN ABOVE THE HVN
                //==================================================

                lvnThreshold =
                     maxVol * vtLvnPct / 100.0

                int lvnRow = na

                if upperHvnRow < vtRows - 1

                    for r = upperHvnRow + 1 to vtRows - 1

                        rowVol =
                             array.get(vtVol, r)

                        if rowVol <= lvnThreshold

                            lvnRow := r
                            break

                //==================================================
                // 6. DETERMINE VOLUME TRIGGER
                //==================================================

                float volTriggerPrice = na

                if not na(lvnRow)

                    // Boundary between the HVN and the first
                    // clearly low-volume area.
                    volTriggerPrice :=
                         vtLow +
                         lvnRow * rowSize

                else

                    // If no clear LVN exists above the HVN,
                    // use the upper edge of the HVN.
                    volTriggerPrice :=
                         vtLow +
                         (upperHvnRow + 1) * rowSize

                //==================================================
                // 7. ONLY SHOW A FUTURE TRIGGER
                //==================================================

                // We don't force the line upward artificially.
                // If the calculated level has already been crossed,
                // it is not useful as a pending GTT.
                if volTriggerPrice > close

                    volTriggerLine := line.new(
                         x1 = bar_index - vtLookback + 1,
                         y1 = volTriggerPrice,
                         x2 = bar_index,
                         y2 = volTriggerPrice,
                         extend = extend.right,
                         color = color.rgb(72, 99, 252),
                         style = line.style_dashed,
                         width = 1
                         )
